cmake_minimum_required(VERSION 3.16)
project(skyshot_cpp VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

file(GLOB SKYSHOT_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
list(REMOVE_ITEM SKYSHOT_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp)

add_library(skyshot STATIC ${SKYSHOT_SOURCES})
target_include_directories(skyshot PUBLIC include PRIVATE third_party)
find_package(Threads REQUIRED)
target_link_libraries(skyshot PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(skyshot PUBLIC /W3 /utf-8 /permissive- /EHsc /fp:precise)
  target_compile_definitions(skyshot PUBLIC _CRT_SECURE_NO_WARNINGS NOMINMAX _SILENCE_CXX20_U8PATH_DEPRECATION_WARNING)
else()
  target_compile_options(skyshot PUBLIC -Wall -Wextra -Wno-unused-parameter -ffp-contract=off)
endif()

add_executable(skyshot_cli src/main.cpp)
set_target_properties(skyshot_cli PROPERTIES OUTPUT_NAME skyshot)
target_link_libraries(skyshot_cli PRIVATE skyshot)

option(SKYSHOT_BUILD_TESTS "Build the test executable" ON)
if(SKYSHOT_BUILD_TESTS)
  file(GLOB SKYSHOT_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp)
  if(SKYSHOT_TEST_SOURCES)
    add_executable(skyshot_tests ${SKYSHOT_TEST_SOURCES})
    target_link_libraries(skyshot_tests PRIVATE skyshot)
    target_compile_definitions(skyshot_tests PRIVATE SKYSHOT_CLI="$<TARGET_FILE:skyshot_cli>")
    add_dependencies(skyshot_tests skyshot_cli)
    enable_testing()
    add_test(NAME skyshot_tests COMMAND skyshot_tests)
  endif()
endif()

option(SKYSHOT_BUILD_EXAMPLES "Build the example programs" ON)
if(SKYSHOT_BUILD_EXAMPLES)
  add_executable(synthetic_demo examples/synthetic_demo.cpp)
  target_link_libraries(synthetic_demo PRIVATE skyshot)
endif()
